cmake_minimum_required(VERSION 3.16)
project(rviz_capture)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(ament_cmake REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rviz_rendering REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rviz_default_plugins REQUIRED)
find_package(rclcpp REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(ament_cmake_python REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "srv/CaptureFrame.srv" DEPENDENCIES builtin_interfaces)
ament_get_python_install_dir(PYTHON_INSTALL_DIR)
install(DIRECTORY python/rviz_capture_client DESTINATION "${PYTHON_INSTALL_DIR}")
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_library(capture_panel SHARED src/capture_panel.cpp src/synced_displays.cpp include/rviz_capture/capture_panel.hpp)
set_target_properties(capture_panel PROPERTIES OUTPUT_NAME rviz_capture)
target_include_directories(capture_panel PRIVATE include)
ament_target_dependencies(capture_panel rviz_common rviz_rendering rviz_default_plugins pluginlib rclcpp)
rosidl_get_typesupport_target(capture_typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(capture_panel "${capture_typesupport}")
target_link_libraries(capture_panel Qt5::Widgets)
target_compile_options(capture_panel PRIVATE -Wall -Wextra -Wpedantic)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS capture_panel LIBRARY DESTINATION lib)
install(PROGRAMS scripts/make_video.py DESTINATION lib/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
ament_package()
